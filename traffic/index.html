机器学习模型训练

RandomForest（R²=0.927）与XGBoost（R²=0.933）对比。10维特征，80/20时序分割。

📊 模型指标对比

🐍 Python 源码

# RandomForest 模型训练
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import r2_score, mean_absolute_error

features = ['temp_c','rain_1h','snow_1h','clouds_all',
            'hour','is_weekend','is_holiday','weather_code',
            'month','weekday']
X = df[features].values
y = df['traffic_volume'].values

# 80/20 时序分割
split = int(len(df) * 0.8)
X_train, X_test = X[:split], X[split:]
y_train, y_test = y[:split], y[split:]

rf = RandomForestRegressor(n_estimators=100, max_depth=15, n_jobs=-1)
rf.fit(X_train, y_train)
y_pred = rf.predict(X_test)

print(f'MAE:  {mean_absolute_error(y_test, y_pred):.1f}')
print(f'R²:   {r2_score(y_test, y_pred):.4f}')
# 输出: MAE: 296.9 | R²: 0.9272

# 高德地图API - 实时路况查询
import requests

def get_traffic_circle(lng, lat):
    url = "https://restapi.amap.com/v3/traffic/status/circle"
    params = {
        "key": "YOUR_AMAP_KEY",
        "location": f"{lng},{lat}",
        "radius": 1000
    }
    r = requests.get(url, params=params)
    data = r.json()
    
    expedite = data['trafficinfo']['evaluation']['expedite']
    congested = data['trafficinfo']['evaluation']['congested']
    print(f"畅通: {expedite}  拥堵: {congested}")
    return data

# 查询上海核心商圈
spots = {
    '人民广场': (121.475, 31.232),
    '陆家嘴':   (121.508, 31.240),
    '静安寺':   (121.448, 31.225),
    '徐家汇':   (121.437, 31.195),
    '五角场':   (121.518, 31.301),
}
for name, (lng, lat) in spots.items():
    get_traffic_circle(lng, lat)
